# --- [ 1. Wake Lock & SSHD ] ---
termux-wake-lock
if ! pgrep -x "sshd" > /dev/null; then sshd; fi

# --- [ 2. ดึงข้อมูลระบบ ] ---
USER_NAME=$(whoami)
IP_ADDR=$(ip addr show | grep 'inet ' | grep -v '127.0.0.1' | awk '{print $2}' | cut -d/ -f1 | head -n 1)

# --- [ 3. Conditional Logic ] ---
if [ -z "$SSH_CONNECTION" ]; then
    # --- กรณีเปิดในแอป Termux บนมือถือโดยตรง ---
    SESSION_COUNT=$(pgrep -c "bash")
    
    if [ "$SESSION_COUNT" -le 1 ]; then
        echo "📱 First Session: Preparing screen 'work1'..."
        
        # ---------------------------------------------------------
        # [🔥 ลบไฟล์ผี กรณีรีบูตเครื่องใหม่ 🔥]
        # เช็คว่ามีโปรแกรม screen รันอยู่จริงๆ ไหม ถ้าไม่มีเลย ให้บังคับลบไฟล์ขยะทิ้ง
        if ! pgrep -x "screen" > /dev/null; then
            echo "🧹 Cleaning up ghost sockets from previous boot..."
            rm -rf ~/.screen/* 2>/dev/null
            rm -rf /data/data/com.termux/files/usr/tmp/screens/* 2>/dev/null
            screen -wipe > /dev/null 2>&1
        fi
        # ---------------------------------------------------------
        
        # เช็คว่ามี session screen ชื่อ work1 ที่ใช้งานได้จริงไหม
        if ! screen -list | grep -q "work1"; then
            echo "🚀 Creating new session and entering Ubuntu..."
            screen -S work1 proot-distro login ubuntu
        else
            echo "🔄 Joining existing session..."
            screen -x work1
        fi
    else
        echo "📱 Additional Session Detected"
        echo "----------------------------------------"
        echo "💻 Windows SSH: ssh $USER_NAME@$IP_ADDR -p 8022"
        echo "----------------------------------------"
        echo "💬 To join screen, type: screen -x work1"
        echo "----------------------------------------"
    fi

else
    # --- กรณี Remote เข้ามาจาก Desktop (SSH Session) ---
    echo "💻 SSH Session: Connected successfully."
    echo "----------------------------------------"
    echo "🛠️  Manage Sessions:"
    echo "1. Sync with Mobile: screen -x work1"
    echo "2. Enter Ubuntu (New): proot-distro login ubuntu"
    echo "----------------------------------------"
fi
